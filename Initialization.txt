// Add to Visitor.cpp
any Visitor::visitInitializedIDDeclaration(CSubsetParser::InitializedIDDeclarationContext *ctx) {
    string name = ctx->ID()->getText();
    symbolTable.Insert(name, "ID", currentDeclType, false, true, false);
    
    bool isGlobalScope = (symbolTable.currentScope->getScopeID() == "1");
    
    if (isGlobalScope) {
        // Note: Standard C only allows constant initialization for globals.
        // If you must support expressions globally, you'd have to move this 
        // code to the start of main(). For now, we assume constant or local.
        globalVars.insert(name);
        // If it's a constant, you could emit: name dd <value>
        // For now, let's treat it as 0 and assign later, or just handle locals.
        emit(name + " dd 1 dup (0)"); 
    } else {
        // 1. Allocate space on stack
        offsetStack.back()[name] = nextLocalOffset;
        nextLocalOffset -= 4;
        totalLocalSize += 4;
        
        // 2. Evaluate RHS (Result goes to EAX)
        visit(ctx->logic_expression());
        
        // 3. Store EAX into the allocated memory
        emit("MOV " + getMemRef(name, offsetStack.back()[name], "") + ", EAX");
    }
    return nullptr;
}

any Visitor::visitMultipleInitializedIDDeclaration(CSubsetParser::MultipleInitializedIDDeclarationContext *ctx) {
    // Handle the previous variables in the list first
    visit(ctx->declaration_list());
    
    string name = ctx->ID()->getText();
    symbolTable.Insert(name, "ID", currentDeclType, false, true, false);
    
    bool isGlobalScope = (symbolTable.currentScope->getScopeID() == "1");
    
    if (!isGlobalScope) {
        // 1. Allocate space
        offsetStack.back()[name] = nextLocalOffset;
        nextLocalOffset -= 4;
        totalLocalSize += 4;
        
        // 2. Evaluate RHS
        visit(ctx->logic_expression());
        
        // 3. Store
        emit("MOV " + getMemRef(name, offsetStack.back()[name], "") + ", EAX");
    }
    return nullptr;
}
