COMPOUND_ASSIGNOP : '+=' | '-=' | '*=' | '/=' | '%=' ; //lexer

expression
    : logic_expression
    | variable ASSIGNOP logic_expression             #AssignExpression
    | variable COMPOUND_ASSIGNOP logic_expression    #CompoundAssignExpression  // <-- ADD THIS
    ;

any Visitor::visitCompoundAssignExpression(CSubsetParser::CompoundAssignExpressionContext *ctx) {
    string op = ctx->COMPOUND_ASSIGNOP()->getText();
    
    // 1. Evaluate RHS (Result goes to EAX)
    visit(ctx->logic_expression());
    emit("PUSH EAX"); // Save RHS value on the stack
    
    // 2. Handle LHS (We only handle simple IDs here for simplicity)
    auto varCtx = ctx->variable();
    if (auto idCtx = dynamic_cast<CSubsetParser::AnIDContext*>(varCtx)) {
        string name = idCtx->ID()->getText();
        int offset = lookupOffset(name);
        string memRef = getMemRef(name, offset, "");
        
        // Load current value of LHS into EAX
        emit("MOV EAX, " + memRef);
        
        // Restore RHS value into EBX
        emit("POP EBX");
        
        // 3. Perform the operation
        if (op == "+=") {
            emit("ADD EAX, EBX");
        } else if (op == "-=") {
            emit("SUB EAX, EBX");
        } else if (op == "*=") {
            emit("IMUL EAX, EBX");
        } else if (op == "/=") {
            emit("CDQ");       // Sign-extend EAX into EDX:EAX for signed division
            emit("IDIV EBX");  // Quotient goes to EAX
        } else if (op == "%=") {
            emit("CDQ");
            emit("IDIV EBX");
            emit("MOV EAX, EDX"); // Remainder goes to EAX
        }
        
        // 4. Store the final result back into the LHS variable
        emit("MOV " + memRef + ", EAX");
    }
    
    return nullptr;
}
